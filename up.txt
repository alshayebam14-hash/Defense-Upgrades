<script>
    let membersData = [];

    function loadData() {
        const url = document.getElementById('scriptUrl').value;
        if(!url) {
            alert('الرجاء إدخال رابط الـ Web App');
            return;
        }

        document.getElementById('loadingArea').style.display = 'block';

        fetch(url + "?sheet=وزارة الدفاع")
            .then(res => res.json())
            .then(data => {
                document.getElementById('loadingArea').style.display = 'none';
                membersData = data;
                renderTable();
            })
            .catch(err => {
                document.getElementById('loadingArea').style.display = 'none';
                alert('حدث خطأ أثناء جلب البيانات: ' + err);
            });
    }

    function renderTable() {
        const tbody = document.getElementById('tableBody');
        tbody.innerHTML = '';

        if(!membersData || membersData.length === 0) {
            tbody.innerHTML = `<tr><td colspan="7">لا توجد بيانات متاحة بالجدول.</td></tr>`;
            return;
        }

        // افتراض هيكلة الصفوف بناءً على جدولك (تعديل حسب الحاجة إذا لزم)
        membersData.forEach((row, index) => {
            if(index === 0) return; // تخطي الهيدر لو وجد
            
            const discordVal = row[3] || '';
            if(!discordVal || discordVal === '<@>') return;

            const name = row[4] || 'بدون اسم';
            const id = row[5] || '';
            const records = row[8] || 0; // افتراض مكان السجلات
            
            // محاكاة الفحص (صح أو خطأ بناءً على متطلبات مبدئية، ويمكن تخصيصها بدقة)
            const isEligible = records >= 5; // شرط افتراضي مؤقت للتجربة

            const tr = document.createElement('tr');
            tr.innerHTML = `
                <td>${discordVal}</td>
                <td>${name}</td>
                <td>جندي</td>
                <td>${records}</td>
                <td>مكتملة</td>
                <td><span class="${isEligible ? 'status-yes' : 'status-no'}">${isEligible ? 'صح (✅)' : 'خطأ (❌)'}</span></td>
                <td>
                    <select onchange="updateStatus(${index}, this.value)">
                        <option value="eligible" ${isEligible ? 'selected' : ''}>صح</option>
                        <option value="not_eligible" ${!isEligible ? 'selected' : ''}>خطأ</option>
                    </select>
                </td>
            `;
            tbody.appendChild(tr);
        });
    }

    function nationalDayUpgrade() {
        alert('تم تفعيل وضع "اليوم الوطني": تم تحويل كافة العساكر إلى مستحقين (صح) بنجاح!');
        // سيتم ربطها بتعديل حالات الجدول محلياً لعرضهم كجاهزين للترقية الفورية
    }

    function previewPromotions() {
        const template = document.getElementById('discordTemplate').value;
        let output = template
            .replace('{userMention}', '<@123456789012345678>\n    • الاسم: أحمد\n    • الكود: GR 💂‍♂️ R-100')
            .replace('{oldRank}', 'جندي')
            .replace('{newRank}', 'جندي أول');

        document.getElementById('previewCard').style.display = 'block';
        document.getElementById('previewOutput').value = output;
    }

    function copyDiscordText() {
        const text = document.getElementById('previewOutput');
        text.select();
        document.execCommand('copy');
        alert('تم نسخ النص بنجاح إلى الحافظة! لصقه مباشرة في ديسكورد ✅');
    }

    function executePromotions() {
        if(!confirm('هل أنت متأكد من اعتماد ترقيات العساكر المحددين ورفعهم في الجدول الأساسي؟')) return;

        const url = document.getElementById('scriptUrl').value;
        const payload = {
            action: "processPromotions",
            promotedUsers: [] // سيتم إرسال القائمة المحددة
        };

        fetch(url, {
            method: 'POST',
            body: JSON.stringify(payload)
        })
        .then(res => res.json())
        .then(res => {
            alert(res.message || 'تمت العملية بنجاح!');
            loadData(); // إعادة تحديث القائمة
        })
        .catch(err => {
            alert('حدث خطأ أثناء تنفيذ الترقيات: ' + err);
        });
    }
</script>
